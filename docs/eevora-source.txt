import { NextRequest, NextResponse } from "next/server";

export const runtime = "nodejs";

const helpLabels: Record<string, string> = {
  "01": "AI Education & Guidance",
  "02": "Business Workflow Discovery",
  "03": "Custom AI Automation",
  "04": "Tool Integration & Workflow Improvement",
  docs: "EevoraDocs",
  unsure: "I’m not sure yet",
};
const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const reply = (message: string, status: number) => NextResponse.json({ message }, { status, headers: { "Cache-Control": "no-store" } });

export async function POST(request: NextRequest) {
  const origin = request.headers.get("origin");
  if (origin) {
    try {
      if (new URL(origin).host !== request.headers.get("host")) return reply("Request origin is not allowed.", 403);
    } catch {
      return reply("Request origin is not allowed.", 403);
    }
  }
  if (!request.headers.get("content-type")?.startsWith("application/json")) return reply("Invalid request format.", 415);
  if (Number(request.headers.get("content-length") || 0) > 12_000) return reply("Your message is too long.", 413);

  let input: Record<string, unknown>;
  try {
    const body = await request.text();
    if (body.length > 12_000) return reply("Your message is too long.", 413);
    const parsed: unknown = JSON.parse(body);
    if (!parsed || typeof parsed !== "object" || Array.isArray(parsed)) return reply("Please check your details and try again.", 400);
    input = parsed as Record<string, unknown>;
  } catch {
    return reply("Please check your details and try again.", 400);
  }

  if (input.website) return NextResponse.json({ ok: true }, { headers: { "Cache-Control": "no-store" } });
  const read = (key: string, max: number) => typeof input[key] === "string" ? (input[key] as string).trim().slice(0, max + 1) : "";
  const name = read("name", 120);
  const business = read("business", 160);
  const email = read("email", 254);
  const help = read("help", 12);
  const challenge = read("challenge", 3000);
  const submissionId = read("submissionId", 80);
  if (!name || name.length > 120 || !business || business.length > 160 || !emailPattern.test(email) || email.length > 254 || !helpLabels[help] || !challenge || challenge.length > 3000 || !/^[a-f0-9-]{36}$/i.test(submissionId)) {
    return reply("Please complete all fields with valid details.", 400);
  }

  const apiKey = process.env.RESEND_API_KEY;
  const to = process.env.CONTACT_TO_EMAIL;
  const from = process.env.CONTACT_FROM_EMAIL;
  if (!apiKey || !to || !from) return reply("Inquiries are temporarily unavailable. Please try again later.", 503);

  try {
    const response = await fetch("https://api.resend.com/emails", {
      method: "POST",
      headers: { Authorization: `Bearer ${apiKey}`, "Content-Type": "application/json", "Idempotency-Key": `eevora-contact/${submissionId}` },
      body: JSON.stringify({
        from,
        to: [to],
        reply_to: email,
        subject: `EEVORA inquiry: ${helpLabels[help]}`,
        text: [`New EEVORA website inquiry`, ``, `Name: ${name}`, `Business or industry: ${business}`, `Email: ${email}`, `Help needed: ${helpLabels[help]}`, ``, `Business challenge:`, challenge].join("\n"),
      }),
      cache: "no-store",
      signal: AbortSignal.timeout(10_000),
    });
    if (!response.ok) {
      console.error("Contact email provider rejected request", response.status);
      return reply("We couldn’t send your inquiry. Please try again later.", 502);
    }
    return NextResponse.json({ ok: true }, { headers: { "Cache-Control": "no-store" } });
  } catch (error) {
    console.error("Contact email delivery failed", error instanceof Error ? error.message : "Unknown error");
    return reply("We couldn’t send your inquiry. Please try again later.", 502);
  }
}
